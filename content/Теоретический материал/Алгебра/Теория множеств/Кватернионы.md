## Кватернионы

### Диаграмма Эйлера-Винна, отображающая [[Бесконечные числовые множества|БЧМ]] с множеством кватернионов. 

```tikz
\usepackage{tikz}
\usetikzlibrary{shapes.geometric}
\usepackage{amssymb}
\begin{document} 
  \begin{tikzpicture} 
    \definecolor{complexcolor}{RGB}{180,50,50}
    \definecolor{realcolor}{RGB}{220,100,100}
    \definecolor{rationalcolor}{RGB}{120,200,120}
    \definecolor{irrationalcolor}{RGB}{150,150,150}
    \definecolor{integercolor}{RGB}{150,150,200}
    \definecolor{naturalcolor}{RGB}{200,200,100}   
    \definecolor{quaternioncolor}{RGB}{100,150,200} % Цвет для кватернионов

    \node [ellipse, minimum width = 18.5cm, minimum height = 7.5cm, fill=quaternioncolor] (H) at (-2,0){}; %H (кватернионы)
    \node [ellipse, minimum width = 15.5cm, minimum height = 7cm, fill=complexcolor] (C) at (-0.5,0){}; %C
    \node [ellipse, minimum width = 12cm, minimum height = 6.5cm, fill=realcolor] (R) at (1,0){}; %R
    \node [ellipse, minimum width = 8cm, minimum height = 5.5cm, fill=rationalcolor] (Q) at (3,0){}; %Q
    \node [ellipse, minimum width = 7cm, minimum height = 4.5cm, fill=integercolor] (Z) at (3.75,0){}; %Z
    \node [ellipse, minimum width = 5.5cm, minimum height = 4cm, fill=naturalcolor] (N) at (4.5,0){}; %N
    \node [ellipse, minimum width = 3.5cm, minimum height = 3cm, fill=irrationalcolor] (I) at (-3,0){}; %I

    
    \node[white] at (-7.25,0) {\huge $\mathbb{C}$}; 
    \node[white] at (-4,0) {\huge $\mathbb{I}$}; 
    \node[white] at (-2,-2) {\huge $\mathbb{R}$}; 
    \node[white] at (-0.25,0) {\huge $\mathbb{Q}$}; 
    \node[white] at (1.25,0) {\huge $\mathbb{Z}$}; 
    \node[white] at (2.5,0) {\huge $\mathbb{N}$}; 
    \node[white] at (-9.5,0) {\huge $\mathbb{H}$}; % Обозначение кватернионов
    
    \node[white] at (-6,1.5) {\Large $3+4i$};
    \node[white] at (-6,0.5) {\Large $2-i$};
    \node[white] at (-6,-1.25) {\Large $-1+0.5i$};
    
    \node[white] at (-2.5,0.5) {\Large $\pi$};
    \node[white] at (-2.5,0) {\Large $e$};
    \node[white] at (-2.5,-0.5) {\Large $\sqrt{3}$};
    
    \node[white] at (0.3,1.5) {\LARGE $\frac{5}{2}$};
    \node[white] at (0.3,-1.5) {\LARGE $-\frac{1}{8}$};
    
    \node[white] at (1.75,1.25) {\Large $-3$};
    \node[white] at (1.5,0.75) {\Large $0$};
    \node[white] at (1.5,-0.75) {\Large $7$};
    
    \node[white] at (4,1) {\Large $5$};
    \node[white] at (5,0) {\Large $10$};
    \node[white ] at (4,-1) {\Large $1$};
    
    \node[white] at (-1.2,2.5) {\Large $\sqrt{2}-1$};
    \node[white] at (-1.2,1.5) {\LARGE $\frac{\sqrt{5}+7}{2}$};
    \node[white] at (-1.2,-1.5) {\LARGE $\frac{\sqrt{3}}{2}$};
    
    % Добавьте примеры кватернионов здесь
    \node[white] at (-9.3,1.2) {\Large $i+j+k$};
    \node[white] at (-9.3,-1.2) {\Large $2i-3j+k$};
  \end{tikzpicture}
\end{document}
```
